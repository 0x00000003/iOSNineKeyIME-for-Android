# Build-host probe, distinct from Android APK and never used as Pixel evidence.
cmake_minimum_required(VERSION 3.18)
project(ime_host_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MODULE_PATH "${CMAKE_SOURCE_DIR}/cmake" ${CMAKE_MODULE_PATH})
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
include(Boost)
set(WITH_GFLAGS OFF CACHE BOOL "" FORCE)
set(WITH_GTEST OFF CACHE BOOL "" FORCE)
set(WITH_UNWIND OFF CACHE BOOL "" FORCE)
add_subdirectory(librime/deps/glog)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(librime/deps/yaml-cpp)
set(SNAPPY_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SNAPPY_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(SNAPPY_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(snappy)
set(HAVE_SNAPPY TRUE)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(librime/deps/leveldb)
add_subdirectory(librime/deps/marisa-trie)
set(LIBMARISA marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
add_subdirectory(OpenCC)
include(OpenccWorkarounds)
set(X11Keysym "${CMAKE_SOURCE_DIR}/librime/include")
include_directories("${X11Keysym}")
# No input-bearing glog output. Host-only build; Android logging separately audited.
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
include(Rime)
add_executable(ime_host_probe probe.cc)
target_include_directories(ime_host_probe PRIVATE librime/src)
target_link_libraries(ime_host_probe PRIVATE rime-static)
add_executable(ime_compatibility_probe compatibility.cc)
target_include_directories(ime_compatibility_probe PRIVATE librime/src "${CMAKE_BINARY_DIR}/librime/src")
target_link_libraries(ime_compatibility_probe PRIVATE rime-static)
add_executable(ime_model_format_probe model_probe.cc)
target_include_directories(ime_model_format_probe PRIVATE librime/src librime-octagram/src "${CMAKE_BINARY_DIR}/librime/src")
target_link_libraries(ime_model_format_probe PRIVATE rime-static)
add_executable(ime_pinyin_probe pinyin_probe.cc)
target_include_directories(ime_pinyin_probe PRIVATE librime/src "${CMAKE_BINARY_DIR}/librime/src")
target_link_libraries(ime_pinyin_probe PRIVATE rime-static)
